                duration = 7300 # 20 years
            }

            # Our overlord shares in the reclamation (conditional effects are hidden in tooltips

                if = {
                    limit = { is_subject = yes }
                    overlord = {

                            duration = 7300 # 20 years
                        }
                    }
                }

            if = {
                limit = { is_subject = no }
                west_india_region = {
                    limit = {
                        NOT = { is_permanent_claim = ROOT }
                        NOT = { is_core = ROOT }
                    }
                    add_permanent_claim = ROOT
                }
            }
            else = {
                overlord = {
                    west_india_region = {
                        limit = {
                            NOT = { is_permanent_claim = PREV }
                            NOT = { is_core = PREV }
                        }
                        add_permanent_claim = PREV
                    }
                }
            }
            custom_tooltip = bts_npl_reclaim_hindustan_subject_claims_tt
        }
