# Nepal content ported from "Beyond the Sunset" (workshop 3735234315). Tibet/Malaya content not ported.

namespace = bts_flavor_npl

country_event = {
    id = bts_flavor_npl.1
    title = "bts_flavor_npl.1.t"
    desc = "bts_flavor_npl.1.d"
    picture = BUDDHA_eventPicture

    is_triggered_only = yes
    major = yes

    option = {
        name = "bts_flavor_npl.1.a"
        ai_chance = {
            factor = 5
            modifier = {
                factor = 0
                religion = vajrayana
            }
        }

        if = {
            limit = {
                religion = hinduism
            }
            add_stability_or_adm_power = yes
        } else = {
            add_stability = -1

            custom_tooltip = bts_newline
            change_religion = hinduism

            hidden_effect = {
                every_owned_province = {
                    limit = {
                        region = hindusthan_region
                        religion = vajrayana
                    }
                    change_religion = ROOT
                }
            }
        }

        custom_tooltip = bts_newline
        add_country_modifier = {
            name = bts_npl_hinduism
            duration = -1
        }

        # Set us down the Hindu mission path
        set_country_flag = bts_npl_unlock_hindu_missions
        swap_non_generic_missions = yes
    }

    option = {
        name = "bts_flavor_npl.2.b"
        ai_chance = {
            factor = 1
        }

        if = {
            limit = {
                religion = vajrayana
            }
            add_prestige_or_monarch_power = {
                amount = 20
            }
        } else = {
            add_stability = -1

            custom_tooltip = bts_newline
            change_religion = vajrayana

            hidden_effect = {
                every_owned_province = {
                    limit = {
                        region = hindusthan_region
                        religion = hinduism
                    }
                    change_religion = ROOT
                }
            }
        }

        custom_tooltip = bts_newline
        add_country_modifier = {
            name = bts_npl_buddhism
            duration = -1
        }
        add_country_modifier = {
            name = bts_warrior_monks
            duration = -1
        }

        # Upgrade Swayambhunath to at least tier 1
        if = {
            limit = {
                557 = {
                    NOT = {
                        has_great_project = {
                            type = swayambhunath
                            tier = 1
                        }
                    }
                }
            }
            557 = {
                add_great_project_tier = {
                    type = swayambhunath
                    tier = 1
                }
            }
        }

        # Set us down the Buddhist mission path
        set_country_flag = bts_npl_unlock_buddhist_missions
        swap_non_generic_missions = yes
    }
}
