country_event = {
	id = mir_glory.2
	title = mir_glory.2.t
	desc = mir_glory.2.d
	picture = REFORM_eventPicture
	is_triggered_only = yes

	option = {
		name = mir_glory.2.a
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0

			limit = { religion = shamanism }

		}
		else = {
			add_prestige = 10
		}
		capital_scope = {
			if = {
				limit = {
					NOT = { religion = coptic }
				}
				change_religion = coptic
			}
		}
	}

	option = {
		name = mir_glory.2.c
